[gcode_macro _FILAMENT_FEED_VARIABLE]
variable_module_sequence: 'left','left','right','right'
variable_channel_sequence: 1,0,0,1
gcode:

[gcode_macro AUTO_FEEDING]
gcode:
  {% set filament_feed_vars = printer['gcode_macro _FILAMENT_FEED_VARIABLE'] | default({}) %}
  {% set extruder = params.EXTRUDER | default(999) | int %}
  {% if filament_feed_vars != {} and
  extruder >= 0 and
  extruder < printer.configfile.settings.printer.max_physical_extruder_num %}
  {% set feed_module_seq = filament_feed_vars.module_sequence[extruder] %}
  {% set feed_channel_seq = filament_feed_vars.channel_sequence[extruder] %}
  FEED_AUTO MODULE={feed_module_seq} CHANNEL={feed_channel_seq} {rawparams}
  {% else %}
  {% if filament_feed_vars == {} %}
  RESPOND TYPE=error MSG='{"gcode_macro _FILAMENT_FEED_VARIABLE error!"}'
  {% else %}
  RESPOND TYPE=error MSG='{"extruder error: %d!" % (extruder)}'
  {% endif %}
  {% endif %}

[gcode_macro MOVE_TO_DISCARD_FILAMENT_POSITION]
gcode:
  {% if printer.toolhead.extruder == 'extruder' %}
  MOVE_TO_XY_IDLE_POSITION_EXTRUDER X_OFFSET=3.5 Y_OFFSET=50.5 {rawparams}
  {% else %}
  MOVE_TO_XY_IDLE_POSITION_EXTRUDER X_OFFSET=-1.5 Y_OFFSET=50.5 {rawparams}
  {% endif %}

[gcode_macro INNER_FLUSH_FILAMENT]
gcode:
  {% set nozzle_diameter = params.NOZZLE_DIAMETER | default(0.4) | float %}
  {% set soft = params.SOFT | default(0) | int %}
  {% set temp = params.TEMP | default(250) | int %}
  {% set speed_default = 360 if soft == 0 else 200 %}
  {% if nozzle_diameter < 0.3 %}
  {% set speed_default = 210 if soft == 0 else 60 %}
  {% endif %}
  {% set speed = params.SPEED | default(speed_default) | int %}
  {% set length = params.LENGTH | default(80.0) | float %}
  {% set retract_length_default = 5.0 if soft == 0 else 2.0 %}
  {% set retract_length = params.RETRACT_LENGTH | default(retract_length_default) | float %}
  {% set e_pos = printer[printer.toolhead.extruder].printing_e_pos | float | abs %}
  {% if printer['print_stats'].state == 'paused' or printer['print_stats'].state == 'printing' %}
  {% set retract_length = e_pos %}
  {% endif %}
  
  M106 S255
  G4 P5000
  INNER_CUTOFF_BASE_DISCARD
  INNER_ROUGHLY_CLEAN_NOZZLE_BASE_DISCARD ACTION=1
  INNER_DISCARD_FILAMENT_BASE_DISCARD
  M107
  M400
  M109 S{temp}
  M83
  G1 E{length} F{speed}
  M400
  M104 S0
  M106 S255
  _CLIENT_RETRACT LENGTH={retract_length}
  M400
  G4 P5000
  INNER_CUTOFF_BASE_DISCARD
  INNER_ROUGHLY_CLEAN_NOZZLE_BASE_DISCARD ACTION=2
  INNER_DISCARD_FILAMENT_BASE_DISCARD
  M107

[gcode_macro INNER_DISCARD_FILAMENT_BASE_DISCARD]
variable_random_pos: 1
gcode:
  {% set accel = params.ACCEL | default(10000) | int %}
  {% set speed = params.SPEED | default(5000) | int %}
  {% set random_position = random_pos % 6 %}
  {% set discard_position = 5.5 + random_position * 0.5 %}
  G91
  M204 S{accel}
  G1 F{speed}
  G1 Y-{discard_position}
  {% if printer.toolhead.extruder == 'extruder' %}
  G1 X-10
  G1 X10
  G1 X-10
  G1 X10
  {% else %}
  G1 X10
  G1 X-10
  G1 X10
  G1 X-10
  {% endif %}
  G1 Y{discard_position}
  SET_GCODE_VARIABLE MACRO=INNER_DISCARD_FILAMENT_BASE_DISCARD VARIABLE=random_pos VALUE={random_pos + 1}

[gcode_macro INNER_PREEXTRUDE_FILAMENT]
gcode:
  {% set temp = params.TEMP | default(220) | int %}
  {% set speed = params.SPEED | default(360) | float %}
  {% set length = params.LENGTH | default(20) | float %}
  {% set retract_length = params.RETRACT_LENGTH | default(0.5) | abs %}
  {% set filament_soft = params.SOFT | default(False) %}
  {% if filament_soft == True %}
  {% set speed = 120 %}
  {% set length = 10 %}
  {% endif %}
  SAVE_GCODE_STATE NAME=preextrude
  SET_ACTION_CODE ACTION=PRINT_PREEXTRUDING
  MOVE_TO_DISCARD_FILAMENT_POSITION
  M204 S10000
  M109 S{temp}
  M83
  INNER_APPLY_FLOW_K EXTRUDER={printer.toolhead.extruder} APPLY=1
  G1 E{length} F{speed}
  INNER_APPLY_FLOW_K EXTRUDER={printer.toolhead.extruder} APPLY=0
  G1 E-{retract_length}
  M400
  M106 S255
  G4 P5000
  M400
  INNER_CUTOFF_BASE_DISCARD
  INNER_ROUGHLY_CLEAN_NOZZLE_BASE_DISCARD ACTION=4
  INNER_DISCARD_FILAMENT_BASE_DISCARD
  MOVE_TO_XY_IDLE_POSITION_EXTRUDER
  M107
  RESTORE_GCODE_STATE NAME=preextrude
  SET_ACTION_CODE ACTION=IDLE

[gcode_macro SM_PRINT_END_AUTO_UNLOAD_FILAMENT]
gcode:
  {% for i in range(printer.configfile.settings.printer.max_physical_extruder_num) %}
  {% if printer.print_task_config['extruders_used'][i] and
  printer.print_task_config['filament_soft'][i] == False and
  printer.print_task_config['end_unload_filament'][i] == True %}
  AUTO_FEEDING EXTRUDER={i} UNLOAD=1 STAGE=prepare
  AUTO_FEEDING EXTRUDER={i} UNLOAD=1 STAGE=doing
  {% endif %}
  {% endfor %}

[gcode_macro PRINT_START]
gcode:
  {% for i in range(printer.configfile.settings.printer.max_physical_extruder_num) %}
  SET_GCODE_VARIABLE MACRO=SM_PRINT_START_LINE_EXTRUDER_{i} VARIABLE=has_finished VALUE=False
  FLOW_APPLY_CALIBRATE_K EXTRUDER={i}
  {% endfor %}
  SET_GCODE_VARIABLE MACRO=AUTO_FEEDING_BATCH VARIABLE=extruder0_temp VALUE=0
  SET_GCODE_VARIABLE MACRO=AUTO_FEEDING_BATCH VARIABLE=extruder1_temp VALUE=0
  SET_GCODE_VARIABLE MACRO=AUTO_FEEDING_BATCH VARIABLE=extruder2_temp VALUE=0
  SET_GCODE_VARIABLE MACRO=AUTO_FEEDING_BATCH VARIABLE=extruder3_temp VALUE=0
  
  M84
  G92 E0
  M220 S100
  {% for section in printer.configfile.config if section.startswith("gcode_macro _PRINT_START_") %}
  { section[12:] }
  {% endfor %}

[gcode_macro PRINT_END]
gcode:
  {% for section in printer.configfile.config if section.startswith("gcode_macro _PRINT_END_") %}
  { section[12:] }
  {% endfor %}
  
  {% set client = printer['gcode_macro _CLIENT_VARIABLE'] | default({}) %}
  {% set retract = client.cancel_retract | default(5.0) | abs %}
  {% set move_z_max_heigh = client.move_z_max_heigh | default(270.5) | abs %}
  
  
  SET_GCODE_VARIABLE MACRO=AUTO_FEEDING_BATCH VARIABLE=doing VALUE=False
  M220 S100
  
  _CLIENT_RETRACT LENGTH={retract}
  
  {% set current_pos = printer.gcode_move.gcode_position %}
  {% set delta_z = 0.0 %}
  {% if "z" in printer.toolhead.homed_axes %}
  {% if current_pos.z < 195 %}
  G90
  G0 Z200 F2000
  {% elif current_pos.z < 265 %}
  G91
  G0 Z5 F2000
  {% else %}
  {% if current_pos.z < move_z_max_heigh - 2.0 %}
  {% set delta_z = 2.0 %}
  {% elif current_pos.z < move_z_max_heigh %}
  {% set delta_z = move_z_max_heigh - current_pos.z %}
  {% else %}
  {% set delta_z = 0.0 %}
  {% endif %}
  G91
  G0 Z{delta_z} F2000
  {% endif %}
  G90
  {% else %}
  RESPOND TYPE=echo MSG='Printer not homed Z'
  {% endif %}
  M400
  INNER_PRINT_END
  {% if "xy" in printer.toolhead.homed_axes %}
  SM_PRINT_END_AUTO_UNLOAD_FILAMENT
  PARK_EXTRUDER
  G90
  G0 X5 Y5 F3000
  {% else %}
  RESPOND TYPE=echo MSG='Printer not homed XY'
  {% endif %}
  
  TURN_OFF_HEATERS
  M107
  M107 P2 S0
  SET_PURIFIER_MODE MODE=0
  {% if printer.print_task_config['end_led_turn_off'] == True %}
  SET_LED LED=cavity_led WHITE=0
  {% endif %}
  SET_PAUSE_NEXT_LAYER ENABLE=0
  SET_PAUSE_AT_LAYER ENABLE=0 LAYER=0
  M84
  SET_GCODE_VARIABLE MACRO=INNER_RESUME VARIABLE=is_pause_on_err VALUE=False
  SET_GCODE_VARIABLE MACRO=INNER_RESUME VARIABLE=pos_e_compensation VALUE=False
  SET_IDLE_TIMEOUT TIMEOUT={printer.configfile.settings.idle_timeout.timeout}
